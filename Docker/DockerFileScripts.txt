DockerFileComSite:


FROM node:latest

RUN apt install git

RUN git clone https://github.com/Lucas-tdo/Way-Safe.git

WORKDIR Way-Safe/Projeto

COPY Way-Safe/Projeto/.env .env
COPY Way-Safe/Projeto/.env.dev .env.dev

RUN npm install

ENV PORT=3333

EXPOSE ${PORT}

CMD ["npm", "start"]
