DockerFileComSite:


FROM node:latest

RUN apt install git

RUN git clone https://github.com/Lucas-tdo/Way-Safe.git

WORKDIR Way-Safe/Projeto

COPY Chaves/.env .env
COPY Chaves/.env.dev .env.dev

RUN npm install

COPY start.sh /usr/local/bin/start.sh
RUN chmod +x /usr/local/bin/start.sh

ENV PORT=3333

EXPOSE ${PORT}

CMD ["bash", "/usr/local/bin/start.sh"]



StartSH:

#!/bin/bash
set -e

cd /Way-Safe/Projeto

echo "🔄 Atualizando repositório..."
