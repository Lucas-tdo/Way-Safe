DockerFileComSite:


FROM node:latest

RUN apt install git

RUN git clone https://github.com/Lucas-tdo/Way-Safe.git

WORKDIR Way-Safe/Projeto

COPY Chaves/.env .env
COPY Chaves/.env.dev .env.dev

RUN npm install

ENV PORT=3333

EXPOSE ${PORT}

CMD ["npm", "start"]
